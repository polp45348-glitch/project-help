ภาษา C ปกติต้อง คอมไพล์ ก่อนรัน คือแปลงโค้ดที่เราเขียน (ไฟล์ .c) ให้เป็นโปรแกรมที่คอมพิวเตอร์เข้าใจ ตัวรันในหน้านี้ทำขั้นตอนนั้นให้แล้ว แต่รองรับแค่คำสั่งพื้นฐาน ถ้าเจอโค้ดที่รันไม่ได้ ให้ลองบนเว็บ onlinegdb.com/online_c_compiler แทน (เปิดในมือถือได้)

มีคอม Windows? ติดตั้ง Dev-C++ ไว้เขียนในเครื่อง
  1. โหลด Embarcadero Dev-C++ จาก github.com/Embarcadero/Dev-Cpp/releases เลือกไฟล์ที่ลงท้ายว่า Setup.exe แล้วติดตั้งตามปกติ (มีคอมไพเลอร์มาให้ในตัว)

  4. กด F11 เพื่อคอมไพล์และรันในทีเดียว จะมีหน้าต่างดำเด้งขึ้นมาแสดงผล

ตัวแปร: กล่องเก็บข้อมูล

จบบทนี้แล้วจะ
  • สร้างตัวแปรเก็บตัวเลขและตัวอักษรได้
  • เอาค่าในตัวแปรไปพิมพ์ด้วย printf
  • รับค่าจากคีย์บอร์ดด้วย scanf และคำนวณเลขได้

ตัวแปรคือกล่องที่มีชื่อ เอาไว้เก็บข้อมูล ในภาษา C ต้องบอกก่อนว่ากล่องนี้เก็บข้อมูลชนิดไหน

ชนิดเก็บอะไรตัวอย่างใน printf ใช้
intจำนวนเต็มint age = 20;%d
floatทศนิยมfloat height = 172.5;%f
charตัวอักษร 1 ตัวchar grade = 'A';%c
vars.c
#include <stdio.h>

int main() {
    int age = 20;
    float height = 172.5;
    char grade = 'A';

    printf("อายุ %d ปี\n", age);
    printf("สูง %.1f ซม.\n", height);   // %.1f = ทศนิยม 1 ตำแหน่ง
    printf("เกรด %c\n", grade);

    age = age + 1;                       // เปลี่ยนค่าในกล่องได้
    printf("ปีหน้าจะอายุ %d ปี\n", age);
    return 0;
}

สูง 172.5 ซม.

ปีหน้าจะอายุ 21 ปี

สังเกต %d ในข้อความเป็นเหมือนช่องว่าง แล้วค่าของตัวแปรที่อยู่หลังเครื่องหมายจุลภาคจะถูกเติมลงไปตามลำดับ

คำนวณเลข

เครื่องหมายความหมายตัวอย่างผล
+ - *บวก ลบ คูณ3 * 412
/หาร7 / 23 (!)
%เศษจากการหาร7 % 21

กับดัก 7 / 2 ได้ 3 ไม่ใช่ 3.5 เพราะ int หาร int ได้ int ส่วนทศนิยมโดนตัดทิ้ง ถ้าอยากได้ 3.5 ให้เขียน 7.0 / 2

รับค่าจากคนใช้งาน

scanf ทำตรงข้ามกับ printf คือรอให้คนพิมพ์ แล้วเอาค่านั้นใส่ตัวแปร อย่าลืมเครื่องหมาย & หน้าชื่อตัวแปร

plus.c
#include <stdio.h>

int main() {
    int a, b;

    printf("ใส่เลขตัวแรก: ");
    scanf("%d", &a);
    printf("ใส่เลขตัวที่สอง: ");
    scanf("%d", &b);

    printf("%d + %d = %d\n", a, b, a + b);
    return 0;
}
ผลลัพธ์ (ตัวเลขคือที่เราพิมพ์เอง)
ใส่เลขตัวแรก: 8
ใส่เลขตัวที่สอง: 5
8 + 5 = 13

ทำไมต้องมี & &a แปลว่า "ที่อยู่ของกล่อง a" scanf ต้องรู้ว่ากล่องอยู่ตรงไหนถึงจะเอาค่าไปใส่ได้ ตอนนี้จำไว้ก่อนว่า scanf ต้องมี & (ยกเว้นตอนรับข้อความ ซึ่งจะเจอทีหลัง)

ตัดสินใจและทำซ้ำ: if กับ loop

จบบทนี้แล้วจะ
  • ให้โปรแกรมเลือกทำต่างกันตามเงื่อนไขด้วย if / else
  • สั่งทำซ้ำหลายรอบด้วย for และ while

if: ถ้า... ให้ทำ...

grade.c
#include <stdio.h>

int main() {
    int score;
    printf("คะแนนสอบ: ");
    scanf("%d", &score);

    if (score >= 80) {
        printf("ได้เกรด A\n");
    } else if (score >= 50) {
        printf("ผ่าน\n");
    } else {
        printf("ไม่ผ่าน สู้ใหม่นะ\n");
    }
    return 0;
}

โปรแกรมจะเช็คเงื่อนไขจากบนลงล่าง เจออันไหนจริงก่อนก็ทำอันนั้นอันเดียวแล้วข้ามที่เหลือ ถ้าไม่จริงสักอัน จะทำใน else

เครื่องหมายความหมาย
==เท่ากับ (สองตัว! ตัวเดียว = คือการใส่ค่า)
!=ไม่เท่ากับ
> < >= <=มากกว่า น้อยกว่า มากกว่าหรือเท่ากับ น้อยกว่าหรือเท่ากับ
&&และ (ต้องจริงทั้งคู่)
||หรือ (จริงอย่างใดอย่างหนึ่งก็พอ)

while: ทำซ้ำจนกว่าเงื่อนไขจะเป็นเท็จ

ใช้ตอนที่ไม่รู้ล่วงหน้าว่าต้องทำกี่รอบ เช่น ให้ทายเลขจนกว่าจะถูก

guess.c
#include <stdio.h>

int main() {
    int secret = 7;
    int guess = 0;

    while (guess != secret) {
        printf("ทายเลข 1-10: ");
        scanf("%d", &guess);
        if (guess < secret) printf("น้อยไป\n");
        if (guess > secret) printf("มากไป\n");
    }
    printf("ถูกต้อง!\n");
    return 0;
}

ระวัง loop ไม่รู้จบ ถ้าเงื่อนไขไม่มีวันเป็นเท็จ โปรแกรมจะวนไปตลอด กด Ctrl + C ใน Terminal เพื่อหยุด

โจทย์ 3.3 (ท้าทาย) สามเหลี่ยมกี่ชั้นก็ได้

กลับไปที่โจทย์ 1.2 คราวนี้ให้รับจำนวนชั้นจากคนใช้งาน ใบ้: ใช้ for ซ้อนกัน 2 ชั้น ตัวนอกนับชั้น ตัวในพิมพ์ *

#include <stdio.h>

int main() {
    int rows;
    printf("กี่ชั้น: ");
    scanf("%d", &rows);
    for (int r = 1; r <= rows; r++) {
        for (int c = 1; c <= r; c++) {
            printf("*");
        }
        printf("\n");
    }
    return 0;
}
